import pandas as pd
import datetime as dt

# 1. Cargar archivo retail
df = pd.read_csv('retail_sales_dataset.csv', parse_dates=['Date'])

# 2. Configurar fecha de análisis
reference_date = df['Date'].max() + dt.timedelta(days=1)

# 3. Agrupación por Cliente (RFM Aggregations)
rfm = df.groupby('Customer ID').agg({
    'Date': lambda x: (reference_date - x.max()).days, # Recency
    'Transaction ID': 'count',                         # Frequency
    'Total Amount': 'sum'                              # Monetary
}).reset_index()

rfm.rename(columns={'Date': 'Recency', 'Transaction ID': 'Frequency', 'Total Amount': 'Monetary'}, inplace=True)

# 4. Cálculo de Quintiles 1 a 5 (qcut)
# Para Recency: menor valor en días es mejor score (5)
rfm['R_Score'] = pd.qcut(rfm['Recency'], 5, labels=[5, 4, 3, 2, 1])
rfm['F_Score'] = pd.qcut(rfm['Frequency'].rank(method='first'), 5, labels=[1, 2, 3, 4, 5])
rfm['M_Score'] = pd.qcut(rfm['Monetary'], 5, labels=[1, 2, 3, 4, 5])

rfm['RFM_Group'] = rfm['R_Score'].astype(str) + rfm['F_Score'].astype(str) + rfm['M_Score'].astype(str)
rfm['RFM_Score'] = rfm[['R_Score', 'F_Score', 'M_Score']].sum(axis=1)

# 5. Mapeo Semántico de Segmentos de Negocio
def assign_segment(row):
    r, f = int(row['R_Score']), int(row['F_Score'])
    if r >= 4 and f >= 4:
        return 'Champions (VIP)'
    elif r >= 3 and f >= 3:
        return 'Leales & Frecuentes'
    elif r >= 4 and f <= 2:
        return 'Nuevos Prometedores'
    elif r == 3 and f <= 2:
        return 'Clientes Potenciales'
    elif r == 2 and f >= 3:
        return 'En Riesgo de Churn'
    elif r <= 2 and f <= 2:
        return 'Hibernando / Perdidos'
    else:
        return 'Requieren Atención'

rfm['Segment'] = rfm.apply(assign_segment, axis=1)
print(rfm.head())
